# videoModel - ASL word recognizer (landmarks, not pixels)
#
# Pinned to the set that was actually resolved, installed and smoke-tested
# together on the machine this project was built on:
#   Windows 11, Python 3.13.15, CPU only (no CUDA), 16 logical CPUs.
#
# WHY NOT numpy<2 (the usual warning for this stack):
#   This machine only has Python 3.13. numpy 1.26.x publishes no cp313 wheels,
#   and mediapipe 1.2.2 / opencv 5.0.0.93 / torch 2.14.1 need numpy>=2. So the
#   pins below are the combination that is proven to work here, including the
#   live webcam path. On Python 3.10/3.11 an older stack (numpy 1.26,
#   mediapipe 0.10.x with mp.solutions) would also work, but nothing in this
#   folder has been tested on it - mediapipe>=1.0 removed the legacy
#   mp.solutions API, and this code uses the Tasks API only.
#
# Install (see README.md, step 1):
#   python -m pip install -r requirements.txt
#
# Do NOT install the streamlit / streamlit-webrtc / av packages from the
# alphabet project (../asl_landmark_app) into this environment. This module is
# self-contained: its webcam demo is an OpenCV window, not a web app.

# --- the four that must match, or the landmark cache is invalid -------------
mediapipe==1.2.2              # HolisticLandmarker (Tasks API, VIDEO mode)
numpy==2.5.3                  # the packed .npy cache is float16 [T, 136, 3]
opencv-python==5.0.0.93       # video decode + the live window and overlay
torch==2.14.1                 # CPU build; models/words_core20.pt is torch.save

# --- training, evaluation, export ------------------------------------------
pandas==3.0.6                 # split CSVs, cache manifest, per-word reports
onnx==1.23.2                  # graph built by python -m src.export_onnx
onnxruntime==1.31.1           # the parity check runs the ONNX graph on CPU

# --- quality of life (the code degrades gracefully without them) -----------
tqdm==4.70.1                  # extraction progress bar
pyarrow==25.0.1               # only if you export the reports as parquet
scikit-learn==1.9.1           # only for extra confusion-matrix reporting
certifi==2026.7.22            # TLS trust store for the one-time model download

# NOT used by this module (deliberately absent):
#   streamlit, streamlit-webrtc, av, matplotlib, seaborn, kagglehub, tensorflow
